def generate_script(theme, tone):
    genai.configure(api_key=GEMINI_API_KEY)
    
    # 404エラーを防ぐため、現在利用可能なモデルを自動検出
    available_models = [
        m.name for m in genai.list_models() 
        if 'generateContent' in m.supported_generation_methods
    ]
    
    # flashモデルがあれば最優先、なければ利用可能な最初のモデルを選択
    selected_model = next(
        (m for m in available_models if 'flash' in m),
        available_models[0] if available_models else 'models/gemini-1.5-flash'
    )
    print(f"使用するGeminiモデル: {selected_model}")
    
    prompt = f"""
    TikTok用のショート動画台本を作成してください。
    
    【テーマ】: {theme}
    【口調・スタイル】: {tone}
    【長さ】: 30秒程度
    
    必ず以下のJSON形式のみを出力してください。マークダウン等の装飾は不要です。
    {{
      "text": "読み上げる台本のテキスト。",
      "keyword": "背景動画を探すための英語の検索キーワード(例: cooking pasta, night city, etc)"
    }}
    """
    model = genai.GenerativeModel(selected_model)
    response = model.generate_content(prompt)
    
    raw_text = response.text.replace("```json", "").replace("```", "").strip()
    return json.loads(raw_text)
